cmake_minimum_required(VERSION 3.16)
project(gaming_mps_core_tests LANGUAGES CXX)
enable_testing()
add_executable(core core.cpp ../common/input.cpp ../common/screen.cpp ../common/code.cpp
    ../common/menu.cpp ../common/stub.cpp ../games/tetris/tetris.cpp
    ../games/minesweeper/minesweeper.cpp ../games/pong/pong.cpp ../games/pong/model.cpp)
target_include_directories(core PRIVATE ../common ../games/tetris ../games/minesweeper ../games/pong)
target_compile_features(core PRIVATE cxx_std_17)
target_compile_options(core PRIVATE -Wall -Wextra -Werror)
add_test(NAME core COMMAND core)
add_executable(pong pong.cpp ../games/pong/model.cpp ../games/pong/pong.cpp ../common/screen.cpp)
target_include_directories(pong PRIVATE ../common ../games/pong)
target_compile_features(pong PRIVATE cxx_std_17)
target_compile_options(pong PRIVATE -Wall -Wextra -Werror)
add_test(NAME pong COMMAND pong)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_FOUND)
  add_test(NAME clipboard COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_clipboard.py")
endif()

add_executable(tetris_rules ../games/tetris/tests/rules.cpp ../games/tetris/tetris.cpp ../common/screen.cpp)
target_include_directories(tetris_rules PRIVATE ../common ../games/tetris)
target_compile_features(tetris_rules PRIVATE cxx_std_17)
target_compile_options(tetris_rules PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME tetris_rules COMMAND tetris_rules)
